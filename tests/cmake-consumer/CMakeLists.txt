cmake_minimum_required(VERSION 3.16)
project(consumer VERSION 91.92.93 LANGUAGES C)

if(CONSUMER_CXX)
    enable_language(CXX)
endif()

# These generic names belong to the application, not its dependencies.
foreach(tool IN ITEMS format format-check cppcheck clang-tidy)
    add_custom_target(${tool})
endforeach()

# Public target suffix and unnamespaced build target pairs from the module contract.
set(expected_names)
foreach(pair IN LISTS EXPECTED_TARGETS)
    string(REGEX MATCH "^([^:]+):(.+)$" matched "${pair}")
    list(APPEND expected_names "${CMAKE_MATCH_1}")
    set(expected_target_${CMAKE_MATCH_1} "${CMAKE_MATCH_2}")
endforeach()
if(NOT "Core" IN_LIST expected_names OR NOT "tlv" IN_LIST expected_names)
    message(FATAL_ERROR "EXPECTED_TARGETS must list the contract targets")
endif()

# Report every unnamespaced target an embedded build creates, recursively.
function(consumer_report_targets directory)
    get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
    foreach(target IN LISTS targets)
        get_target_property(type ${target} TYPE)
        file(APPEND "${CMAKE_BINARY_DIR}/opentlv-embedded.txt" "target ${target} ${type}\n")
    endforeach()
    get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
    foreach(child IN LISTS children)
        consumer_report_targets("${child}")
    endforeach()
endfunction()

set(source_mode OFF)
if(CONSUMER_MODE MATCHES "^package")
    set(components)
    if(CONSUMER_MODE STREQUAL "package")
        set(components COMPONENTS Core OPTIONAL_COMPONENTS unavailable)
    endif()
    find_package(OpenTLV ${EXPECTED_VERSION} EXACT CONFIG REQUIRED ${components})
    if(OpenTLV_unavailable_FOUND)
        message(FATAL_ERROR "Unknown optional component was reported as available")
    endif()
else()
    get_cmake_property(cache_before CACHE_VARIABLES)
    if(CONSUMER_MODE STREQUAL "subdirectory")
        set(source_mode ON)
        add_subdirectory("${CONSUMER_SOURCE_DIR}" opentlv)
        set(library_source "${CONSUMER_SOURCE_DIR}")
    else()
        include(FetchContent)
        set(package_args)
        if(NOT CMAKE_VERSION VERSION_LESS 3.24)
            set(package_args FIND_PACKAGE_ARGS ${EXPECTED_VERSION} EXACT CONFIG NAMES OpenTLV)
        elseif(CONSUMER_MODE STREQUAL "fetch-package")
            message(FATAL_ERROR "FIND_PACKAGE_ARGS requires CMake 3.24")
        endif()
        # Neither source override nor installed discovery may attempt this download.
        FetchContent_Declare(${CONSUMER_DEPENDENCY}
            GIT_REPOSITORY "${CMAKE_CURRENT_SOURCE_DIR}/nonexistent-repository"
            GIT_TAG unused
            ${package_args})
        FetchContent_MakeAvailable(${CONSUMER_DEPENDENCY})
        if(NOT CONSUMER_MODE STREQUAL "fetch-package")
            set(source_mode ON)
            set(library_source "${${CONSUMER_DEPENDENCY}_SOURCE_DIR}")
        endif()
    endif()
    if(source_mode)
        get_cmake_property(cache_after CACHE_VARIABLES)
        file(WRITE "${CMAKE_BINARY_DIR}/opentlv-embedded.txt" "")
        consumer_report_targets("${library_source}")
        foreach(snapshot IN ITEMS before after)
            foreach(entry IN LISTS cache_${snapshot})
                file(APPEND "${CMAKE_BINARY_DIR}/opentlv-embedded.txt" "${snapshot} ${entry}\n")
            endforeach()
        endforeach()
    endif()
endif()

# The same contract names resolve in every mode: aliases of the unnamespaced
# build targets from source, imported targets from an installed package.
foreach(name IN LISTS expected_names)
    # COMPONENTS Core imports Core, checked below, and no other contract target.
    if(CONSUMER_MODE STREQUAL "package" AND NOT name STREQUAL "Core")
        if(TARGET OpenTLV::${name})
            message(FATAL_ERROR "COMPONENTS Core also loaded OpenTLV::${name}")
        endif()
        continue()
    endif()
    if(NOT TARGET OpenTLV::${name})
        message(FATAL_ERROR "Missing contract target OpenTLV::${name}")
    endif()
    get_target_property(imported OpenTLV::${name} IMPORTED)
    get_target_property(aliased OpenTLV::${name} ALIASED_TARGET)
    if(source_mode AND (imported OR NOT aliased STREQUAL expected_target_${name}))
        message(FATAL_ERROR "OpenTLV::${name} must alias ${expected_target_${name}}, not ${aliased}")
    elseif(NOT source_mode AND NOT imported)
        message(FATAL_ERROR "OpenTLV::${name} was not imported from the installed package")
    endif()
    if(CONSUMER_MODE STREQUAL "package-all" AND NOT OpenTLV_${name}_FOUND)
        message(FATAL_ERROR "OpenTLV_${name}_FOUND is not set by unrestricted discovery")
    endif()
endforeach()

# CMake 3.24+ redirects find_package(OpenTLV) to a FetchContent dependency only
# when its name is the lowercase package name, the contract's canonical `opentlv`.
if(CONSUMER_MODE STREQUAL "fetch-canonical" AND NOT CMAKE_VERSION VERSION_LESS 3.24)
    find_package(OpenTLV ${EXPECTED_VERSION} EXACT CONFIG REQUIRED)
    if(NOT OpenTLV_DIR STREQUAL CMAKE_FIND_PACKAGE_REDIRECTS_DIR)
        message(FATAL_ERROR "find_package(OpenTLV) bypassed the embedded Library: ${OpenTLV_DIR}")
    endif()
endif()

if(NOT PROJECT_NAME STREQUAL "consumer" OR NOT PROJECT_VERSION STREQUAL "91.92.93")
    message(FATAL_ERROR "Core changed the consumer's project identity")
endif()
if(NOT TARGET OpenTLV::Core OR TARGET OpenTLV::tlvpp)
    message(FATAL_ERROR "Incorrect public CMake targets")
endif()
if(TARGET OpenTLV::tlv++ OR TARGET tlv++)
    message(FATAL_ERROR "Obsolete C++ target is still exported")
endif()
if(TARGET otlv OR TARGET nlohmann_json::nlohmann_json)
    message(FATAL_ERROR "Core must not provide CLI targets or its JSON dependency")
endif()
get_target_property(core_imported OpenTLV::Core IMPORTED)
if(source_mode)
    if(core_imported OR OPENTLV_BUILD_TESTS OR OPENTLV_BUILD_EXAMPLES)
        message(FATAL_ERROR "Source override must embed only the Core libraries by default")
    endif()
elseif(NOT core_imported)
    message(FATAL_ERROR "Package discovery did not select the installed library")
else()
    get_target_property(core_location OpenTLV::Core IMPORTED_LOCATION_RELEASE)
    file(RELATIVE_PATH relative_location "${EXPECTED_PREFIX}" "${core_location}")
    if(IS_ABSOLUTE "${relative_location}" OR relative_location MATCHES "^\\.\\.")
        message(FATAL_ERROR "Core was not loaded from the relocated installation: ${core_location}")
    endif()
endif()

enable_testing()
add_executable(consumer_c main.c)
target_link_libraries(consumer_c PRIVATE OpenTLV::Core)
set(consumers consumer_c)
if(CONSUMER_CXX)
    add_executable(consumer_cpp main.cpp)
    # No explicit link to C, include directories, or language standard.
    target_link_libraries(consumer_cpp PRIVATE OpenTLV::Core)
    list(APPEND consumers consumer_cpp)
endif()
get_target_property(core_type OpenTLV::Core TYPE)
foreach(consumer IN LISTS consumers)
    if(WIN32 AND core_type STREQUAL "SHARED_LIBRARY")
        add_custom_command(TARGET ${consumer} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "$<TARGET_FILE:OpenTLV::Core>"
                "$<TARGET_FILE_DIR:${consumer}>"
            VERBATIM)
    endif()
    add_test(NAME ${consumer} COMMAND ${consumer} "${EXPECTED_RUNTIME_VERSION}")
endforeach()
