cmake_minimum_required(VERSION 3.15)
project(OpenTLVConsumer LANGUAGES ${CONSUMER_LANGUAGE})
include(../strict_compile_options.cmake)

set(CMAKE_DISABLE_FIND_PACKAGE_GTest TRUE)
set(CONSUMPTION_MODE "find_package" CACHE STRING "How the consumer obtains OpenTLV")
if(CONSUMPTION_MODE STREQUAL "find_package")
    # Restrict both lookups to the candidate, even on a host with OpenTLV installed.
    find_package(OpenTLV 0.0.9 QUIET CONFIG PATHS "${PACKAGE_PREFIX}" NO_DEFAULT_PATH)
    if(OpenTLV_FOUND)
        message(FATAL_ERROR "The package accepted an incompatible request for OpenTLV 0.0.9")
    endif()
    find_package(OpenTLV 0.1.0 EXACT CONFIG REQUIRED PATHS "${PACKAGE_PREFIX}" NO_DEFAULT_PATH)
    if(NOT OpenTLV_VERSION STREQUAL "0.1.0")
        message(FATAL_ERROR "Expected OpenTLV 0.1.0, found '${OpenTLV_VERSION}'")
    endif()
    set(core_dependency OpenTLV::core)
elseif(CONSUMPTION_MODE STREQUAL "add_subdirectory")
    add_subdirectory("${OPENTLV_SOURCE_DIR}" opentlv)
    set(core_dependency opentlv_core)
elseif(CONSUMPTION_MODE STREQUAL "FetchContent")
    include(FetchContent)
    FetchContent_Declare(opentlv SOURCE_DIR "${OPENTLV_SOURCE_DIR}")
    FetchContent_MakeAvailable(opentlv)
    set(core_dependency opentlv_core)
else()
    message(FATAL_ERROR "Unknown consumption mode '${CONSUMPTION_MODE}'")
endif()
if(NOT CONSUMPTION_MODE STREQUAL "find_package")
    get_directory_property(inherited_testing DIRECTORY "${OPENTLV_SOURCE_DIR}" DEFINITION BUILD_TESTING)
    if(NOT "${inherited_testing}" STREQUAL "${BUILD_TESTING}")
        message(FATAL_ERROR "OpenTLV changed the parent's BUILD_TESTING setting in its directory")
    endif()
endif()
if(TARGET opentlv_core_tests OR TARGET opentlv_io_tests OR TARGET opentlv_integration_tests)
    message(FATAL_ERROR "Consuming OpenTLV enabled its test suites")
endif()

get_target_property(core_links OpenTLV::core INTERFACE_LINK_LIBRARIES)
get_target_property(io_links OpenTLV::io INTERFACE_LINK_LIBRARIES)
if(core_links MATCHES "(OpenTLV::io|opentlv_io)" OR NOT core_dependency IN_LIST io_links)
    message(FATAL_ERROR "Expected IO to depend publicly on Core and Core to be usable alone. "
                        "Core links: ${core_links}. IO links: ${io_links}.")
endif()

enable_testing()
set(CMAKE_${CONSUMER_LANGUAGE}_STANDARD_REQUIRED YES)
set(CMAKE_${CONSUMER_LANGUAGE}_EXTENSIONS OFF)
if(CONSUMER_LANGUAGE STREQUAL "C")
    set(CMAKE_C_STANDARD 99)
    set(probe_extension c)
    add_executable(core-consumer consumer.c)
    add_executable(io-consumer ../../examples/write.c)
else()
    # The public targets must raise this project's default to C++17.
    set(CMAKE_CXX_STANDARD 14)
    set(probe_extension cpp)
    add_executable(core-consumer core.cpp)
    add_executable(io-consumer consumer.cpp)
    add_subdirectory(../abi abi)
endif()

foreach(module IN ITEMS core io)
    if(DEFINED EXPECTED_LIBRARY_TYPE)
        get_target_property(library_type OpenTLV::${module} TYPE)
        if(NOT library_type STREQUAL EXPECTED_LIBRARY_TYPE)
            message(FATAL_ERROR "Expected OpenTLV::${module} to be ${EXPECTED_LIBRARY_TYPE}, "
                                "found ${library_type}")
        endif()
    endif()
    target_link_libraries(${module}-consumer PRIVATE OpenTLV::${module})
    opentlv_strict_compile_options(${module}-consumer ${CONSUMER_LANGUAGE})
    add_test(NAME ${module}-consumer COMMAND ${module}-consumer)
    if(CONSUMER_LANGUAGE STREQUAL "CXX" AND DOC_EXAMPLES_DIR)
        add_executable(doc-${module} "${DOC_EXAMPLES_DIR}/${module}.cpp")
        target_link_libraries(doc-${module} PRIVATE OpenTLV::${module})
        opentlv_strict_compile_options(doc-${module} CXX)
        add_test(NAME doc-${module} COMMAND doc-${module})
    endif()
endforeach()

# Use the inventory only for header names. The compiler resolves every include
# through the linked target. The driver also checks the installed inventory.
set(HEADER_INVENTORY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../include" CACHE PATH "Public headers to probe")
set(include_root "${HEADER_INVENTORY_ROOT}")
file(GLOB_RECURSE headers RELATIVE "${include_root}" "${include_root}/opentlv/*.h")
if(CONSUMER_LANGUAGE STREQUAL "CXX")
    file(GLOB_RECURSE cpp_headers RELATIVE "${include_root}" "${include_root}/opentlv/*.hpp")
    list(APPEND headers ${cpp_headers})
endif()
foreach(header IN LISTS headers)
    if(header MATCHES "^opentlv/core/")
        set(module core)
    else()
        # The umbrella headers expose both modules.
        set(module io)
    endif()
    string(REPLACE "/" "_" probe "${header}")
    set(source "${CMAKE_CURRENT_BINARY_DIR}/${probe}.${probe_extension}")
    file(WRITE "${source}" "#include <${header}>\ntypedef int header_probe;\n")
    target_sources(${module}-consumer PRIVATE "${source}")
endforeach()
